$page = max(1, (int)($_GET['page'] ?? 1));
$perPage = 20;

$sql = "SELECT p.*, 
        p2.name as parent_name,
        u.fullname as created_by_name,
        (SELECT COUNT(*) FROM bom WHERE product_id = p.id) as parts_count
        FROM products p
        LEFT JOIN products p2 ON p2.id = p.parent_product_id
        LEFT JOIN users u ON u.id = p.created_by
        WHERE 1=1";
$params = [];

if ($status) {
    $sql .= " AND p.status = :status";
    $params[':status'] = $status;
}

if ($search) {
    $sql .= " AND (p.code LIKE :search OR p.name LIKE :search OR p.type LIKE :search)";
    $params[':search'] = '%' . $search . '%';
}

$sql .= " ORDER BY p.created_at DESC";

$result = db()->paginate($sql, $params, $page, $perPage);
$products = $result['data'];
